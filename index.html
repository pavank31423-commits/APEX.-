        function renderCurrentQuestion() {

            const qObj = activeQuestions[currentQIndex];

            document.getElementById('qCounter').textContent = `Question ${currentQIndex + 1} of ${activeQuestions.length}`;

            document.getElementById('qCategoryBadge').textContent = qObj.sub;

            document.getElementById('qText').textContent = `Q${currentQIndex + 1}. ${qObj.q}`;

            const labels = ['A', 'B', 'C', 'D'];

            document.getElementById('qOptions').innerHTML = qObj.opts.map((opt, idx) => {

                const isSelected = userAnswers[currentQIndex] === idx;

                const btnStyle = isSelected

                    ? "bg-amber-400 text-slate-950 border-amber-300 font-bold"

                    : "bg-white/10 hover:bg-white/20 text-white border-white/15";

                return `

                    <button onclick="selectOption(${idx})" class="${btnStyle} border p-3.5 rounded-xl text-left text-xs sm:text-sm transition flex items-center gap-3">

                        <span class="w-6 h-6 rounded-lg bg-black/20 flex items-center justify-center font-bold text-xs">${labels[idx]}</span>

                        <span>${opt}</span>

                    </button>`;

            }).join('');

        }

        function submitCBTTest() {

            let correct = 0, wrong = 0, skipped = 0;

            let solHTML = "";

            activeQuestions.forEach((q, idx) => {

                const uAns = userAnswers[idx];

                if (uAns === undefined) {

                    skipped++;

                    solHTML += `<div class="p-2.5 rounded-lg bg-slate-900 border border-slate-700"><b>Q${idx+1}: ${q.q}</b><br><span class="text-amber-400">छोड़ा गया (Skipped)</span> | सही उत्तर: <b>${q.opts[q.ans]}</b><br><span class="text-slate-400">${q.exp}</span></div>`;

                } else if (uAns === q.ans) {

                    correct++;

                    solHTML += `<div class="p-2.5 rounded-lg bg-emerald-950/50 border border-emerald-500/30"><b>Q${idx+1}: ${q.q}</b><br><span class="text-emerald-400">✅ सही जवाब (+4): ${q.opts[q.ans]}</span><br><span class="text-slate-300">${q.exp}</span></div>`;

                } else {

                    wrong++;

                    solHTML += `<div class="p-2.5 rounded-lg bg-rose-950/50 border border-rose-500/30"><b>Q${idx+1}: ${q.q}</b><br><span class="text-rose-400">❌ गलत (-1): आपने चुना ${q.opts[uAns]}</span> | सही उत्तर: <b>${q.opts[q.ans]}</b><br><span class="text-slate-300">${q.exp}</span></div>`;

                }

            });

            const totalScore = (correct * 4) - (wrong * 1);

            const maxScore = activeQuestions.length * 4;

            lastScoreText = `मैंने APEX STUDY के लाइव टेस्ट में ${maxScore} में से ${totalScore} अंक हासिल किए (${correct} सही, ${wrong} गलत)! तुम भी अपना टेस्ट देकर देखो: ${window.location.href}`;

            document.getElementById('resScore').textContent = `${totalScore} / ${maxScore}`;

            document.getElementById('resCorrect').textContent = correct;

            document.getElementById('resWrong').textContent = wrong;

            document.getElementById('resSkipped').textContent = skipped;

            document.getElementById('resultSummaryText').textContent = `आपने ${activeQuestions.length} में से ${correct + wrong} सवाल हल किए।`;

            document.getElementById('resSolutions').innerHTML = solHTML;

            document.getElementById('quizPlayArea').classList.add('hidden');

            document.getElementById('quizResultArea').classList.remove('hidden');

        }
